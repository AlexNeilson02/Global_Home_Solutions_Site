function getRedirectUrl(role: string): string {
  switch (role) {
    case "admin":
      return "/admin-portal";
    case "contractor":
      return "/contractor-portal";
    case "salesperson":
      return "/sales-portal";
    case "homeowner":
      return "/homeowner-portal";
    default:
      return "/";
  }
}
Step 5: Create API Route for Homeowner Bids
In server/routes.ts, add this route to fetch homeowner's bid requests:

// Get bid requests for a specific homeowner
app.get("/api/homeowner/:userId/bids", isAuthenticated, async (req, res) => {
  try {
    const userId = parseInt(req.params.userId);
    const authenticatedUserId = (req.user as any).id;
    
    // Ensure users can only access their own bids
    if (userId !== authenticatedUserId) {
      return res.status(403).json({ message: "Forbidden" });
    }
    
    const bidRequests = await storage.getBidRequestsByUserId(userId);
    res.json({ bidRequests });
  } catch (error) {
    console.error("Error fetching homeowner bid requests:", error);
    res.status(500).json({ message: "Failed to fetch bid requests" });
  }
});

async getBidRequestsByUserId(userId: number): Promise<BidRequest[]> {
  return db
    .select()
    .from(bidRequests)
    .where(eq(bidRequests.userId, userId))
    .orderBy(desc(bidRequests.createdAt));
}
